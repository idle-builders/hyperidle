/**
 * P12 — prototype of the R2 predicate: "a service value must not reach a layer face's
 * export surface". Built on the parser the live census already uses
 * (`scripts/lib/repo-census.ts:644-646`: `ts.createSourceFile`).
 *
 * WHAT THE RULE READS: the module's top-level EXPORT DECLARATIONS, and what those
 * declarations say. Not what later statements do to an already-exported object.
 *
 * The whole rule, in a paragraph:
 *
 *   In a layer root package's modules reachable from its `exports` map, an import from a
 *   service package may not reach the module's export surface. Four export forms carry a
 *   value: `export … from "<service>"` (named clause, `export *`, `export * as ns`),
 *   `export { local }` with no specifier, `export default <expr>`, and an exported
 *   `const`/`let`/`var`. The exported expression is read through object and array
 *   literals, spreads, parens, `as`/`satisfies`, and call arguments, and through
 *   top-level `const`/`let`/`var` aliases (transitively). A service symbol found anywhere
 *   in that reach is a violation. One exemption, positional: a bare service identifier in
 *   direct callee position of a call or `new` — calling a service factory PRODUCES a
 *   value, and producing values is what composition is.
 *
 * WHAT IT DOES NOT READ, deliberately (founder ruling, concept gate: lint is engineering
 * guidance, not protection against a determined author):
 *   - function-like bodies (arrows, functions, methods, accessors, constructors) — the
 *     sanctioned wiring style IS a bag of arrows over the layer's own services, so a body
 *     is machine-indistinguishable from legitimate composition;
 *   - class/enum/namespace members;
 *   - statements that mutate an already-exported object (`api.tool = svc` and its
 *     variants), aliases, globals, dynamic `import()`.
 *
 * A developer who wants to route a service symbol out through the face can. That is
 * expected of a lint rule and is not a defect: this rule exists so the plain re-export
 * and deps-bag spellings — the ones written by accident — fail the build.
 *
 * Usage: npx tsx r2-escape-prototype.ts <file...>
 * Output: per file, `VIOLATION line N — <binding> (from <specifier>) — <context>`, or CLEAN.
 */

import { readFileSync } from "node:fs";
import ts from "typescript";

const FRAMEWORK_PACKAGES = new Set(["shared", "api-contract", "db", "design-system"]);
const LAYER_PACKAGES = new Set(["core-api", "agent-orchestration", "agent-execution", "core-ui"]);

function isServiceSpecifier(spec: string): boolean {
	if (!spec.startsWith("@hyperidle/")) return false;
	const name = spec.slice("@hyperidle/".length).split("/")[0] ?? "";
	return !FRAMEWORK_PACKAGES.has(name) && !LAYER_PACKAGES.has(name);
}

interface Finding {
	readonly line: number;
	readonly binding: string;
	readonly specifier: string;
	readonly context: string;
}

function analyze(filePath: string): Finding[] {
	const source = readFileSync(filePath, "utf8");
	const sf = ts.createSourceFile(filePath, source, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
	const findings: Finding[] = [];

	function lineOf(node: ts.Node): number {
		return sf.getLineAndCharacterOfPosition(node.getStart(sf)).line + 1;
	}
	function isExported(node: ts.Node): boolean {
		const mods = ts.canHaveModifiers(node) ? ts.getModifiers(node) : undefined;
		return (mods ?? []).some((m) => m.kind === ts.SyntaxKind.ExportKeyword);
	}
	function unwrapParens(e: ts.Expression): ts.Expression {
		let cur = e;
		while (ts.isParenthesizedExpression(cur)) cur = cur.expression;
		return cur;
	}

	/** Service value imports: local name -> specifier. Reaching one is the violation. */
	const serviceBindings = new Map<string, string>();
	/** Top-level value declarations: name -> the expressions that feed it. */
	const aliases = new Map<string, ts.Expression[]>();

	function addAlias(name: string, expr: ts.Expression | undefined): void {
		const list = aliases.get(name) ?? [];
		if (expr) list.push(expr);
		aliases.set(name, list);
	}
	function patternNames(pattern: ts.BindingName, out: string[]): void {
		if (ts.isIdentifier(pattern)) {
			out.push(pattern.text);
			return;
		}
		for (const el of pattern.elements) {
			if (!ts.isOmittedExpression(el)) patternNames(el.name, out);
		}
	}

	// ---- pass 1: what this module holds — service imports and top-level aliases ----
	for (const stmt of sf.statements) {
		if (ts.isImportDeclaration(stmt)) {
			if (!ts.isStringLiteral(stmt.moduleSpecifier)) continue;
			const spec = stmt.moduleSpecifier.text;
			if (!isServiceSpecifier(spec)) continue;
			const clause = stmt.importClause;
			if (!clause || clause.isTypeOnly) continue; // type-only: no value escapes
			if (clause.name) serviceBindings.set(clause.name.text, spec);
			if (clause.namedBindings) {
				if (ts.isNamespaceImport(clause.namedBindings)) {
					serviceBindings.set(clause.namedBindings.name.text, spec);
				} else {
					for (const el of clause.namedBindings.elements) {
						if (!el.isTypeOnly) serviceBindings.set(el.name.text, spec);
					}
				}
			}
		} else if (ts.isVariableStatement(stmt)) {
			for (const decl of stmt.declarationList.declarations) {
				if (ts.isIdentifier(decl.name)) {
					addAlias(decl.name.text, decl.initializer);
				} else {
					const names: string[] = [];
					patternNames(decl.name, names);
					for (const n of names) addAlias(n, decl.initializer);
				}
			}
		}
	}

	// ---- reading an escaping expression ----
	const seenNodes = new Set<ts.Node>();
	const seenNames = new Set<string>();

	/** Every value in this expression reaches the export surface. */
	function readEscaping(node: ts.Node, context: string): void {
		if (seenNodes.has(node)) return;
		seenNodes.add(node);

		if (ts.isIdentifier(node)) {
			readEscapingName(node.text, node, context);
			return;
		}
		if (ts.isCallExpression(node) || ts.isNewExpression(node)) {
			const callee = unwrapParens(node.expression);
			// THE ONE EXEMPTION, positional: a bare service identifier in direct callee
			// position. The call's RESULT is a produced value; the symbol does not flow into it.
			if (!(ts.isIdentifier(callee) && serviceBindings.has(callee.text))) {
				readEscaping(callee, `${context} > callee`);
			}
			for (const arg of node.arguments ?? []) readEscaping(arg, `${context} > call argument`);
			return;
		}
		if (ts.isObjectLiteralExpression(node)) {
			for (const prop of node.properties) {
				if (ts.isPropertyAssignment(prop)) readEscaping(prop.initializer, context);
				else if (ts.isShorthandPropertyAssignment(prop)) readEscaping(prop.name, context);
				else if (ts.isSpreadAssignment(prop)) readEscaping(prop.expression, `${context} > spread`);
				// methods, getters, setters: function-like bodies — not read
			}
			return;
		}
		// function-like values, classes, enums, namespaces: own-module code — not read
		if (
			ts.isArrowFunction(node) ||
			ts.isFunctionExpression(node) ||
			ts.isFunctionDeclaration(node) ||
			ts.isClassLike(node) ||
			ts.isEnumDeclaration(node) ||
			ts.isModuleDeclaration(node) ||
			ts.isTypeNode(node)
		) {
			return;
		}
		// array literals, spreads, template spans, conditionals, binaries, await,
		// as/satisfies, parens, property access: read through generically
		ts.forEachChild(node, (child) => readEscaping(child, context));
	}

	/** A name reaches the export surface: flag a service import, else follow its aliases. */
	function readEscapingName(name: string, at: ts.Node, context: string): void {
		const spec = serviceBindings.get(name);
		if (spec) {
			findings.push({ line: lineOf(at), binding: name, specifier: spec, context });
			return;
		}
		if (seenNames.has(name)) return;
		seenNames.add(name);
		for (const expr of aliases.get(name) ?? []) {
			readEscaping(expr, `${context} > \`${name}\``);
		}
	}

	// ---- pass 2: the export forms ----
	for (const stmt of sf.statements) {
		if (ts.isExportDeclaration(stmt)) {
			if (stmt.isTypeOnly) continue;
			if (stmt.moduleSpecifier && ts.isStringLiteral(stmt.moduleSpecifier)) {
				// `export … from` / `export * from` / `export * as ns from` a service module
				const spec = stmt.moduleSpecifier.text;
				if (isServiceSpecifier(spec)) {
					findings.push({
						line: lineOf(stmt),
						binding: stmt.exportClause ? "(re-export)" : "*",
						specifier: spec,
						context: "export … from a service module",
					});
				}
			} else if (stmt.exportClause && ts.isNamedExports(stmt.exportClause)) {
				// `export { local }` — the local name is the escape
				for (const el of stmt.exportClause.elements) {
					if (el.isTypeOnly) continue;
					const local = el.propertyName ?? el.name;
					readEscapingName(local.text, el, `named export of \`${local.text}\``);
				}
			}
		} else if (ts.isExportAssignment(stmt)) {
			readEscaping(stmt.expression, "default export");
		} else if (ts.isVariableStatement(stmt) && isExported(stmt)) {
			for (const decl of stmt.declarationList.declarations) {
				if (ts.isIdentifier(decl.name)) {
					readEscapingName(decl.name.text, decl.name, "exported binding");
				} else if (decl.initializer) {
					readEscaping(decl.initializer, "exported destructured binding");
				}
			}
		}
	}

	return findings.sort((a, b) => a.line - b.line);
}

let anyFinding = false;
for (const file of process.argv.slice(2)) {
	const findings = analyze(file);
	console.log(`== ${file}`);
	if (findings.length === 0) {
		console.log("   CLEAN — no service value reaches the export surface");
	} else {
		anyFinding = true;
		for (const f of findings) {
			console.log(`   VIOLATION line ${f.line} — \`${f.binding}\` (from ${f.specifier}) — ${f.context}`);
		}
	}
}
process.exit(anyFinding ? 1 : 0);

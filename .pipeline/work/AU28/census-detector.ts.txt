const sfCache = new Map<string, ts.SourceFile>();
function parse(path: string): ts.SourceFile {
	let sf = sfCache.get(path);
	if (sf === undefined) {
		sf = ts.createSourceFile(path, readFile(path), ts.ScriptTarget.Latest, true, path.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS);
		sfCache.set(path, sf);
	}
	return sf;
}
const lineOf = (sf: ts.SourceFile, n: ts.Node): number => sf.getLineAndCharacterOfPosition(n.getStart(sf)).line + 1;

type Surface =
	| "db:table"
	| "db:row-type"
	| "db:schema-ns"
	| "db:client"
	| "db:bootstrap"
	| "db:util"
	| "db:testing"
	| "db:unknown";

const TABLES = new Set<string>();
const ROW_TYPES = new Set<string>();
{
	const schema = parse("framework/db/src/schema.ts");
	for (const st of schema.statements) {
		const exported = ts.canHaveModifiers(st) && ts.getModifiers(st)?.some((m) => m.kind === ts.SyntaxKind.ExportKeyword) === true;
		if (!exported) continue;
		if (ts.isVariableStatement(st)) {
			for (const d of st.declarationList.declarations) {
				if (!ts.isIdentifier(d.name)) continue;
				const init = d.initializer;
				if (init !== undefined && ts.isCallExpression(init) && ts.isIdentifier(init.expression) && init.expression.text === "pgTable") {
					TABLES.add(d.name.text);
				}
			}
		} else if (ts.isTypeAliasDeclaration(st) && /\$infer(Select|Insert)/.test(st.type.getText(schema))) {
			ROW_TYPES.add(st.name.text);
		}
	}
}

/**
 * Module → surface. The three symbol lists in Ruling 11.2 (`db:client`, `db:bootstrap`, `db:util`)
 * each land wholly inside one source module, so the mapping is stated once per module rather than
 * per symbol. `system-prompt-templates.ts` is deliberately absent — Ruling 11.2 does not name it and
 * guessing a surface for it would be fabrication, so its symbols fall to `db:unknown` and are
 * reported.
 */
const MODULE_SURFACE: Record<string, Surface> = {
	"connection.ts": "db:client",
	"migrate.ts": "db:client",
	"write-executor.ts": "db:client",
	"migrate-pg.ts": "db:client",
	"bootstrap-pg.ts": "db:client",

	"factory-bootstrap.ts": "db:bootstrap",

/**
 * `@hyperidle/db/zod-schemas` symbol → the TABLE it is generated from.
 *
 * Round-1 answer 6: every export is `createSelectSchema(<table>)` / `createInsertSchema(<table>)`, so
 * the module is global but every SYMBOL is per-table. A schema generated from a table is the same
 * access wearing a different hat — allowing the barrel while forbidding the table import would be a
 * hole around the table ACL. So these resolve to `db:table` and inherit the owner rule; the surface
 * `db:zod-schemas` no longer exists as a separately-ruled thing.
 */
const ZOD_TABLE = new Map<string, string>();
{
	const zs = parse("framework/db/src/zod-schemas.ts");
	for (const st of zs.statements) {
		if (!ts.isVariableStatement(st)) continue;
		for (const d of st.declarationList.declarations) {
			if (!ts.isIdentifier(d.name) || d.initializer === undefined || !ts.isCallExpression(d.initializer)) continue;
			const fn = d.initializer.expression;
			if (!ts.isIdentifier(fn) || !/^create(Select|Insert)Schema$/.test(fn.text)) continue;
			const a = d.initializer.arguments[0];
			if (a !== undefined && ts.isIdentifier(a) && TABLES.has(a.text)) ZOD_TABLE.set(d.name.text, a.text);
		}
	}
}

/** Barrel symbol → surface, built from `framework/db/src/index.ts`'s re-export list. */
const BARREL_SURFACE = new Map<string, Surface>();
const UNKNOWN_SYMBOLS = new Map<string, string>(); // symbol → why
{
	const index = parse("framework/db/src/index.ts");
	for (const st of index.statements) {
		if (!ts.isExportDeclaration(st) || st.moduleSpecifier === undefined || !ts.isStringLiteral(st.moduleSpecifier)) continue;
		const mod = st.moduleSpecifier.text.replace(/^\.\//, "").replace(/\.js$/, ".ts");
		const ec = st.exportClause;
		if (ec === undefined) continue;
		if (ts.isNamespaceExport(ec)) {
			BARREL_SURFACE.set(ec.name.text, "db:schema-ns");
			continue;
		}
		for (const el of ec.elements) {
			const sym = el.name.text;
			if (mod === "schema.ts") {
				if (TABLES.has(sym)) BARREL_SURFACE.set(sym, "db:table");
				else if (ROW_TYPES.has(sym)) BARREL_SURFACE.set(sym, "db:row-type");
				else {
					BARREL_SURFACE.set(sym, "db:unknown");
					UNKNOWN_SYMBOLS.set(sym, "exported from schema.ts but neither a pgTable nor a $inferSelect/$inferInsert row type");
				}
				continue;
			}
			const surf = MODULE_SURFACE[mod];
			if (surf === undefined) {
				BARREL_SURFACE.set(sym, "db:unknown");
				UNKNOWN_SYMBOLS.set(sym, `re-exported from ./${mod}, a module Ruling 11.2 does not name`);
			} else BARREL_SURFACE.set(sym, surf);
		}
	}
}

/** Surface for a db import: resolved target FILE first, then the imported SYMBOL. */
/** For a zod-schema symbol, the table it is generated from; otherwise the symbol itself. */
const tableForSymbol = (toFile: string, sym: string): string => (toFile.endsWith("zod-schemas.ts") ? (ZOD_TABLE.get(sym) ?? sym) : sym);

function surfaceForSymbol(toFile: string, sym: string, isNamespace: boolean): Surface {
	const base = toFile.slice("framework/db/src/".length);
	if (base.startsWith("__testing__/") || base.startsWith("__tests__/")) return "db:testing";
	if (base === "zod-schemas.ts") return ZOD_TABLE.has(sym) ? "db:table" : "db:unknown";
	if (base === "schema.ts") {
		if (isNamespace) return "db:schema-ns";
		if (TABLES.has(sym)) return "db:table";
		if (ROW_TYPES.has(sym)) return "db:row-type";
		UNKNOWN_SYMBOLS.set(sym, "imported from schema.ts directly but neither a pgTable nor a row type");
		return "db:unknown";
	}
	if (base === "index.ts") {
		if (isNamespace) return "db:schema-ns";
		const s = BARREL_SURFACE.get(sym);
		if (s !== undefined) return s;
		UNKNOWN_SYMBOLS.set(sym, "not present in framework/db/src/index.ts's re-export list");
		return "db:unknown";
	}
	const s = MODULE_SURFACE[base];
	if (s !== undefined) return s;
	UNKNOWN_SYMBOLS.set(`${base}:${sym}`, `deep import of framework/db/src/${base}, a module Ruling 11.2 does not name`);
	return "db:unknown";
}

/**
 * READ / WRITE SPLIT OF THE TABLE KIND. `table` used to be one kind carrying both halves of database
 * access, which made the rule row `<role> → db:table (other) [table]` unable to say the thing the
 * founder decided: in the development world a foreign-table READ is allowed and a foreign-table
 * WRITE is not. The kind is therefore split at the instance, driven by the Drizzle operation already
 * detected in `Inst.op`:
 *
 *   table:read   `from` (select), `leftJoin`/`innerJoin`/`rightJoin`/`fullJoin`, and a bare
 *                schema-namespace dereference (`schema.users` outside a Drizzle call — a column
 *                reference in a where/order clause, which reads and never writes).
 *   table:write  `insert`, `update`, `delete`. Drizzle spells an upsert `insert(...).onConflictDo…`,
 *                so an upsert is already counted as an insert and lands here.
 *
 * A TABLE-SYMBOL IMPORT IS NEITHER. `import { users } from "@hyperidle/db"` obtains the symbol; it
 * performs no database operation at all, so it stays on the `[import]` row where it already was.
 * Reading the import row as "a read" would double-count the access and mis-state the rule.
 */
type Kind = "import" | "manifest" | "di" | "api" | "mcp" | "event" | "table:read" | "table:write" | "capability";

/** Drizzle ops that mutate. Everything else on a table is a read. */
const WRITE_OPS = new Set(["insert", "update", "delete", "upsert"]);
const tableKind = (op: string): Kind => (WRITE_OPS.has(op) ? "table:write" : "table:read");

interface Inst {
	kind: Kind;
	source: string;
	target: string;
	file: string;
	line: number;
	symbol: string;
	op: string;
	world: "production" | "development";
	prodImportsDev?: boolean;
	/** table symbol, when the target is a table */
	table?: string;
}
const insts: Inst[] = [];
const add = (i: Inst): void => {
	insts.push({ ...i, file: normPath(i.file) });
};

function litText(sf: ts.SourceFile, a: ts.Expression | undefined): string {
	if (a === undefined) return "(none)";
	if (ts.isStringLiteralLike(a)) return a.text;
	if (ts.isTemplateExpression(a)) {
		let out = a.head.text;
		for (const sp of a.templateSpans) out += `\${…}${sp.literal.text}`;
		return out;
	}
	return `(dynamic:${a.getText(sf).replace(/\s+/g, " ").slice(0, 40)})`;
}

const TS_FILES = files.filter((f) => /\.(m|c)?tsx?$/.test(f) && !f.endsWith(".d.ts"));

/**
 * Exported VALUE CONSTANTS, per package: `export const X = [...] as const`, a frozen array, or a
 * primitive. Round-1 answer 10: all four production `layer → framework [di]` instances were
 * `z.enum(enumValues(BET_OUTCOME_VERDICTS))` and friends — a frozen array imported from `shared` and
 * passed inline to a Zod schema. No port, no injection, no inversion. Passing a constant is an
 * ordinary import, so the DI detector must not score it, or that row carries permanent noise.
 *
 * Deliberately narrow: only array literals and primitives count. An object literal is NOT treated as
 * a constant, because an object of functions is a plausible injected port and suppressing it would
 * hide a real crossing.
 */
const CONST_EXPORTS = new Map<string, Set<string>>();
{
	const unwrap = (e: ts.Expression): ts.Expression => {
		let x = e;
		for (;;) {
			if (ts.isAsExpression(x) || ts.isParenthesizedExpression(x) || ts.isTypeAssertionExpression(x)) x = x.expression;
			else if (ts.isCallExpression(x) && ts.isPropertyAccessExpression(x.expression) && x.expression.name.text === "freeze" && x.arguments[0] !== undefined) x = x.arguments[0] as ts.Expression;
			else return x;
		}
	};
	const isConstValue = (e: ts.Expression): boolean => {
		const x = unwrap(e);
		return (
			ts.isArrayLiteralExpression(x) ||
			ts.isStringLiteralLike(x) ||
			ts.isNumericLiteral(x) ||
			x.kind === ts.SyntaxKind.TrueKeyword ||
			x.kind === ts.SyntaxKind.FalseKeyword
		);
	};
	for (const path of TS_FILES) {
		const owner = pkgFor(path).short;
		const sf = parse(path);
		for (const st of sf.statements) {
			if (!ts.isVariableStatement(st)) continue;
			if (ts.getModifiers(st)?.some((m) => m.kind === ts.SyntaxKind.ExportKeyword) !== true) continue;
			for (const d of st.declarationList.declarations) {
				if (!ts.isIdentifier(d.name) || d.initializer === undefined) continue;
				if (!isConstValue(d.initializer)) continue;
				const set = CONST_EXPORTS.get(owner) ?? new Set<string>();
				set.add(d.name.text);
				CONST_EXPORTS.set(owner, set);
			}
		}
	}
}
let diConstantsSuppressed = 0;

/**
 * Which Layer serves which `/api/...` path. Built by AST from every route registration inside a
 * Layer — `.route(p, …)`, `.use(p, …)`, `.get/.post/.put/.patch/.delete/.all(p, …)` — so it moves
 * with the code instead of being a hand list.
 *
 * Round-1 answer 3: `application → layer [api]` read 0 because the census counted only `apiFetch()`,
 * `ws*.subscribe()` and seam-port calls. `applications/remote-worker` is a DETACHED PROCESS — the one
 * application that cannot compose Layers in-process — and dials both Layers with a raw `fetch` and a
 * raw WebSocket. Longest-prefix match, because both `/ws` (core-api) and `/ws/worker`
 * (agent-orchestration) exist and only the longer one is right.
 */
const ROUTE_INDEX: { path: string; layer: LayerName }[] = [];
const ROUTE_FNS = new Set(["route", "use", "get", "post", "put", "patch", "delete", "all", "on"]);
for (const path of TS_FILES) {
	const owner = pkgFor(path);
	if (owner.layer === null || kindOf(path) !== "production") continue;
	const sf = parse(path);
	const walk = (n: ts.Node): void => {
		// A route registration always passes a path AND at least one handler. Without the arity test,
		// `someMap.get("/x")` and `set.delete("/x")` land in the index and mis-attribute real dials —
		// which is exactly how `/api/mcp/` first resolved to the wrong Layer.
		if (ts.isCallExpression(n) && ts.isPropertyAccessExpression(n.expression) && ROUTE_FNS.has(n.expression.name.text) && n.arguments.length >= 2) {
			const a = n.arguments[0];
			if (a !== undefined && ts.isStringLiteral(a) && a.text.startsWith("/") && a.text.length > 1) {
				ROUTE_INDEX.push({ path: a.text.replace(/\/$/, ""), layer: owner.layer as LayerName });
			}
		}
		ts.forEachChild(n, walk);
	};
	walk(sf);
}
ROUTE_INDEX.sort((a, b) => b.path.length - a.path.length);
/**
 * Resolve a dialled `/api/...` path to the Layer that DECLARES it. Null when nothing matches.
 *
 * Segment-wise, because two things break a plain string prefix: a registered `:param` matches any
 * one segment (`/mcp/:token`), and a dialled url whose tail is interpolated stops early
 * (`${base}/api/mcp/${token}` yields `/api/mcp/`). Longest match wins — that is what tells core-api's
 * `/ws` apart from agent-orchestration's `/ws/worker`.
 */
function layerServing(apiPath: string): LayerName | null {
	const want = apiPath
		.replace(/^\/api/, "")
		.replace(/[?#].*$/, "")
		.split("/")
		.filter((x) => x !== "");
	if (want.length === 0) return null;
	let best: { n: number; layer: LayerName } | null = null;
	for (const r of ROUTE_INDEX) {
		const have = r.path.split("/").filter((x) => x !== "");
		// The FIRST segment must match literally. Without this a route registered as `/:id` matches any
		// url on earth — including `https://openrouter.ai/api/v1/auth/key`, which is not our API at all.
		if (have.length === 0 || (have[0] as string).startsWith(":") || have[0] !== want[0]) continue;
		const n = Math.min(have.length, want.length);
		let ok = true;
		for (let k = 0; k < n; k += 1) {
			const h = have[k] as string;
			if (h.startsWith(":")) continue;
			if (h !== want[k]) {
				ok = false;
				break;
			}
		}
		if (ok && (best === null || n > best.n)) best = { n, layer: r.layer };
	}
	return best === null ? null : best.layer;
}
const shortOfLayerName = (l: LayerName): string => packages.find((q) => q.role === "layer" && q.layer === l)?.short ?? l;

// --- 1. import (from the reused edge set), db split per SYMBOL ---------------

/**
 * The surfaces a single db import statement reaches. A db import is decomposed into one instance per
 * SURFACE (and, for tables, per TABLE symbol) because the rule differs per surface and per table
 * owner — one instance per statement could not carry that. Non-db edges stay one instance each.
 */
function dbSurfacesForEdge(from: string, specifier: string, to: string): { surface: Surface; symbol: string; table?: string }[] {
	const out: { surface: Surface; symbol: string; table?: string }[] = [];
	const seen = new Set<string>();
	const push = (surface: Surface, symbol: string): void => {
		const table = surface === "db:table" ? tableForSymbol(to, symbol) : undefined;
		const key = surface === "db:table" ? `${surface}:${table as string}` : surface;
		if (seen.has(key)) return;
		seen.add(key);
		out.push({ surface, symbol, ...(table === undefined ? {} : { table }) });
	};
	const sf = parse(from);
	const walk = (node: ts.Node): void => {
		const spec =
			(ts.isImportDeclaration(node) || ts.isExportDeclaration(node)) && node.moduleSpecifier !== undefined && ts.isStringLiteral(node.moduleSpecifier)
				? node.moduleSpecifier.text
				: null;
		if (spec === specifier) {
			if (ts.isImportDeclaration(node)) {
				const cl = node.importClause;
				if (cl === undefined) push("db:unknown", "(side-effect import, no binding)");
				else {
					if (cl.name !== undefined) push(surfaceForSymbol(to, cl.name.text, false), cl.name.text);
					const nb = cl.namedBindings;
					if (nb !== undefined) {
						if (ts.isNamespaceImport(nb)) push(surfaceForSymbol(to, nb.name.text, true), `* as ${nb.name.text}`);
						else for (const el of nb.elements) push(surfaceForSymbol(to, (el.propertyName ?? el.name).text, false), (el.propertyName ?? el.name).text);
					}
				}
			}
		}
		// `const { createPgliteDb } = await import("@hyperidle/db")` — round-1 answer 5: two of the
		// three production `db:unknown` instances were this shape, not an unclassified surface. The
		// AST can read the binding pattern; it just was not being asked to.
		if (ts.isCallExpression(node) && node.expression.kind === ts.SyntaxKind.ImportKeyword) {
			const a = node.arguments[0];
			if (a !== undefined && ts.isStringLiteralLike(a) && a.text === specifier) {
				let n: ts.Node = node;
				while (n.parent !== undefined && (ts.isAwaitExpression(n.parent) || ts.isParenthesizedExpression(n.parent) || ts.isAsExpression(n.parent))) n = n.parent;
				const decl = n.parent;
				if (decl !== undefined && ts.isVariableDeclaration(decl) && ts.isObjectBindingPattern(decl.name)) {
					for (const el of decl.name.elements) {
						const sym = (el.propertyName !== undefined && ts.isIdentifier(el.propertyName) ? el.propertyName : el.name) as ts.Identifier;
						if (ts.isIdentifier(sym)) push(surfaceForSymbol(to, sym.text, false), sym.text);
					}
				} else if (decl !== undefined && ts.isVariableDeclaration(decl) && ts.isIdentifier(decl.name)) {
					push("db:schema-ns", `await import() bound whole as ${decl.name.text}`);
				} else {
					push("db:unknown", "await import() whose bindings the AST cannot read");
				}
			}
		}
		if (spec === specifier) {
			if (ts.isExportDeclaration(node)) {
				const ec = node.exportClause;
				if (ec === undefined) push("db:schema-ns", "export * (re-export of the whole module)");
				else if (ts.isNamespaceExport(ec)) push("db:schema-ns", `export * as ${ec.name.text}`);
				else for (const el of ec.elements) push(surfaceForSymbol(to, (el.propertyName ?? el.name).text, false), (el.propertyName ?? el.name).text);
			}
		}
		ts.forEachChild(node, walk);
	};
	walk(sf);
	if (out.length === 0) push("db:unknown", "(specifier not resolvable to an import clause)");
	return out;
}

let dbEdges = 0;
let dbSymbolInstances = 0;
for (const e of census.edges) {
	const s = pkgFor(e.from);
	const t = pkgFor(e.to);
	const fromKind = kindOf(e.from);
	const toKind = kindOf(e.to);
	const prodImportsDev = fromKind === "production" && toKind === "development";
	const base = {
		kind: "import" as const,
		source: s.short,
		file: e.from,
		line: e.line,
		world: fromKind,
		...(prodImportsDev ? { prodImportsDev: true } : {}),
	};
	if (prodImportsDev) {
		add({ ...base, target: `${t.short} [development file]`, symbol: e.specifier, op: "import→development-file" });
		continue;
	}
	if (t.short === "db" && s.short !== "db" && /\.(m|c)?tsx?$/.test(e.from)) {
		dbEdges += 1;
		for (const { surface, symbol, table } of dbSurfacesForEdge(e.from, e.specifier, e.to)) {
			dbSymbolInstances += 1;
			add({
				...base,
				target: surface === "db:table" ? `db:table:${table ?? symbol}` : surface,
				symbol,
				op: symbol.startsWith("await import") ? "dynamic-import" : "import",
				...(surface === "db:table" ? { table: table ?? symbol } : {}),
			});
		}
		continue;
	}
	add({ ...base, target: t.short, symbol: e.specifier, op: "import" });
}

const EVENTS = new Set<string>();
for (const m of readFile("framework/shared/src/events.ts").matchAll(/type:\s*"([a-z0-9_]+)"/g)) EVENTS.add(m[1] as string);

const LLM_SDK = [/^openai$/, /^@anthropic-ai\/sdk/, /^@google\/generative-ai/];
const MCP_SDK = [/^@modelcontextprotocol\/sdk/];
const HARNESS_BINS = new Set(["claude", "codex", "opencode", "cursor-agent", "gh"]);
const SPAWN_FNS = new Set(["spawn", "spawnSync", "exec", "execSync", "execFile", "execFileSync", "fork"]);

interface EvSite {
	pkg: Pkg;
	file: string;
	line: number;
	event: string;
	dir: "emit" | "on";
	world: "production" | "development";
}
const evSites: EvSite[] = [];
let dynamicEventOnSites = 0;
let rejectedEventLiterals = 0;

const DIRECT_OPS: Record<string, string> = { from: "read", insert: "insert", update: "update", delete: "delete" };
const JOIN_OPS = new Set(["leftJoin", "innerJoin", "rightJoin", "fullJoin"]);

function importedIdentifierOrigins(path: string, sf: ts.SourceFile): Map<string, Pkg> {
	const out = new Map<string, Pkg>();
	const rel = edgesByFrom.get(path);
	for (const st of sf.statements) {
		if (!ts.isImportDeclaration(st) || !ts.isStringLiteral(st.moduleSpecifier)) continue;
		if (st.importClause?.isTypeOnly === true) continue;
		const spec = st.moduleSpecifier.text;
		let tgt: Pkg | undefined;
		if (spec.startsWith("@hyperidle/")) tgt = byName.get(spec.split("/").slice(0, 2).join("/"));
		else if (spec.startsWith(".")) {
			const to = rel?.get(spec);
			if (to !== undefined) tgt = pkgFor(to);
		}
		if (tgt === undefined) continue;
		const cl = st.importClause;
		if (cl === undefined) continue;
		if (cl.name !== undefined) out.set(cl.name.text, tgt);
		const nb = cl.namedBindings;
		if (nb !== undefined) {
			if (ts.isNamespaceImport(nb)) out.set(nb.name.text, tgt);
			else for (const el of nb.elements) if (el.isTypeOnly !== true) out.set(el.name.text, tgt);
		}
	}
	return out;
}

for (const path of TS_FILES) {
	const p = pkgFor(path);
	const world = kindOf(path);
	const sf = parse(path);
	const origins = importedIdentifierOrigins(path, sf);

	const dbNs = new Set<string>();
	for (const st of sf.statements) {
		if (
			ts.isImportDeclaration(st) &&
			ts.isStringLiteral(st.moduleSpecifier) &&
			(st.moduleSpecifier.text.startsWith("@hyperidle/db") || (edgesByFrom.get(path)?.get(st.moduleSpecifier.text) ?? "").startsWith("framework/db/")) &&
			st.importClause?.namedBindings !== undefined &&
			ts.isNamespaceImport(st.importClause.namedBindings)
		) {
			dbNs.add(st.importClause.namedBindings.name.text);
		}
	}

	const portRoots = new Map<string, "core-api" | "agent-orchestration">();
	const typeKind = (t: string): "core-api" | "agent-orchestration" | null =>
		/AgentOrchestration\w*Port|buildAgentOrchestrationPort/.test(t) ? "agent-orchestration" : /CoreApi\w*Port|buildCoreApiPort/.test(t) ? "core-api" : null;
	const collectRoots = (n: ts.Node): void => {
		if ((ts.isVariableDeclaration(n) || ts.isParameter(n) || ts.isPropertyDeclaration(n)) && ts.isIdentifier(n.name)) {
			const fromType = n.type !== undefined ? typeKind(n.type.getText(sf)) : null;
			const init = (n as { initializer?: ts.Expression }).initializer;
			const fromInit = init !== undefined && ts.isCallExpression(init) && ts.isIdentifier(init.expression) ? typeKind(init.expression.text) : null;
			const k = fromType ?? fromInit;
			if (k !== null) portRoots.set(n.name.text, k);
		}
		ts.forEachChild(n, collectRoots);
	};
	collectRoots(sf);

	const visit = (n: ts.Node): void => {
		// table — Drizzle ops
		if (ts.isCallExpression(n) && ts.isPropertyAccessExpression(n.expression)) {
			const m = n.expression.name.text;
			const op = DIRECT_OPS[m] ?? (JOIN_OPS.has(m) ? "join" : null);
			if (op !== null) {
				const a = n.arguments[0];
				let tbl: string | null = null;
				if (a !== undefined && ts.isIdentifier(a) && TABLES.has(a.text)) tbl = a.text;
				else if (a !== undefined && ts.isPropertyAccessExpression(a) && ts.isIdentifier(a.expression) && dbNs.has(a.expression.text) && TABLES.has(a.name.text))
					tbl = a.name.text;
				if (tbl !== null)
					add({ kind: tableKind(op), source: p.short, target: `db:table:${tbl}`, table: tbl, file: path, line: lineOf(sf, n.expression.name), symbol: tbl, op, world });
			}
		}
		// table — schema-namespace dereference
		if (ts.isPropertyAccessExpression(n) && ts.isIdentifier(n.expression) && dbNs.has(n.expression.text) && TABLES.has(n.name.text)) {
			const parentIsDrizzleArg =
				n.parent !== undefined &&
				ts.isCallExpression(n.parent) &&
				ts.isPropertyAccessExpression(n.parent.expression) &&
				(DIRECT_OPS[n.parent.expression.name.text] !== undefined || JOIN_OPS.has(n.parent.expression.name.text));
			if (!parentIsDrizzleArg)
				add({
						kind: tableKind("schema-ns-deref"),
						source: p.short,
						target: `db:table:${n.name.text}`,
						table: n.name.text,
						file: path,
						line: lineOf(sf, n),
						symbol: n.name.text,
						op: "schema-ns-deref",
						world,
					});
		}
		// capability — SDK imports
		if ((ts.isImportDeclaration(n) || ts.isExportDeclaration(n)) && n.moduleSpecifier !== undefined && ts.isStringLiteral(n.moduleSpecifier)) {
			const spec = n.moduleSpecifier.text;
			if (LLM_SDK.some((r) => r.test(spec))) add({ kind: "capability", source: p.short, target: "capability:llm-sdk", file: path, line: lineOf(sf, n), symbol: spec, op: "import", world });
			else if (MCP_SDK.some((r) => r.test(spec))) add({ kind: "capability", source: p.short, target: "capability:mcp-sdk", file: path, line: lineOf(sf, n), symbol: spec, op: "import", world });
		}
		if (ts.isCallExpression(n) && n.expression.kind === ts.SyntaxKind.ImportKeyword) {
			const a = n.arguments[0];
			if (a !== undefined && ts.isStringLiteralLike(a)) {
				if (LLM_SDK.some((r) => r.test(a.text)))
					add({ kind: "capability", source: p.short, target: "capability:llm-sdk", file: path, line: lineOf(sf, n), symbol: a.text, op: "dynamic-import", world });
				else if (MCP_SDK.some((r) => r.test(a.text)))
					add({ kind: "capability", source: p.short, target: "capability:mcp-sdk", file: path, line: lineOf(sf, n), symbol: a.text, op: "dynamic-import", world });
			}
		}
		if (ts.isCallExpression(n)) {
			const callee = n.expression;
			const fname = ts.isIdentifier(callee) ? callee.text : ts.isPropertyAccessExpression(callee) ? callee.name.text : "";
			const baseText = ts.isPropertyAccessExpression(callee) ? callee.expression.getText(sf) : "";
			if (SPAWN_FNS.has(fname)) {
				const a = n.arguments[0];
				if (a !== undefined && ts.isStringLiteralLike(a)) {
					const bin = a.text.split("/").pop() ?? a.text;
					if (HARNESS_BINS.has(bin)) add({ kind: "capability", source: p.short, target: "capability:harness-cli", file: path, line: lineOf(sf, n), symbol: bin, op: `${fname}()`, world });
				}
			}
			if (ts.isPropertyAccessExpression(callee) && callee.name.text === "emit") {
				const a = n.arguments[0];
				if (a !== undefined && ts.isObjectLiteralExpression(a)) {
					const tp = a.properties.find((pr): pr is ts.PropertyAssignment => ts.isPropertyAssignment(pr) && ts.isIdentifier(pr.name) && pr.name.text === "type");
					if (tp !== undefined && ts.isStringLiteralLike(tp.initializer)) {
						if (EVENTS.has(tp.initializer.text)) evSites.push({ pkg: p, file: path, line: lineOf(sf, callee.name), event: tp.initializer.text, dir: "emit", world });
						else rejectedEventLiterals += 1;
					}
				}
			}
			if (ts.isPropertyAccessExpression(callee) && callee.name.text === "on") {
				const a = n.arguments[0];
				if (a !== undefined && ts.isStringLiteralLike(a)) {
					if (a.text === "*" || EVENTS.has(a.text)) evSites.push({ pkg: p, file: path, line: lineOf(sf, callee.name), event: a.text, dir: "on", world });
					else rejectedEventLiterals += 1;
				} else if (a !== undefined && ts.isIdentifier(a)) dynamicEventOnSites += 1;
			}
			if (ts.isIdentifier(callee) && callee.text === "apiFetch") {
				// (the seam helper — raw dials are handled below, on the literal itself)
			}
			if (ts.isIdentifier(callee) && callee.text === "apiFetch") {
				add({ kind: "api", source: p.short, target: "core-api", file: path, line: lineOf(sf, n), symbol: litText(sf, n.arguments[0]), op: "apiFetch", world });
			}
			if (ts.isPropertyAccessExpression(callee) && callee.name.text === "subscribe" && /\bws|wsClient|socket/i.test(callee.expression.getText(sf))) {
				add({ kind: "api", source: p.short, target: "core-api", file: path, line: lineOf(sf, n), symbol: litText(sf, n.arguments[0]), op: "ws-subscribe", world });
			}
			if (ts.isPropertyAccessExpression(callee)) {
				const chain: string[] = [];
				let cur: ts.Expression = callee;
				while (ts.isPropertyAccessExpression(cur)) {
					chain.unshift(cur.name.text);
					cur = cur.expression;
				}
				let k: "core-api" | "agent-orchestration" | null = null;
				if (ts.isIdentifier(cur)) k = portRoots.get(cur.text) ?? null;
				else if (ts.isCallExpression(cur) && ts.isIdentifier(cur.expression)) k = typeKind(cur.expression.text);
				if (k !== null && chain.length >= 1) add({ kind: "api", source: p.short, target: k, file: path, line: lineOf(sf, n), symbol: chain.join("."), op: "seam-port", world });
			}
			if (fname === "createMcpToolBridge") add({ kind: "mcp", source: p.short, target: "mcp-router", file: path, line: lineOf(sf, n), symbol: fname, op: "bridge-construct", world });
			if (fname === "registerTool") add({ kind: "mcp", source: p.short, target: "mcp-router", file: path, line: lineOf(sf, n), symbol: litText(sf, n.arguments[0]), op: "sdk-register-tool", world });
			if (fname === "open" && /mcp/i.test(baseText)) add({ kind: "mcp", source: p.short, target: "mcp-router", file: path, line: lineOf(sf, n), symbol: `${baseText}.open`, op: "session-open", world });
			if (origins.size > 0) {
				const calleeRoot = ts.isIdentifier(callee) ? callee.text : ts.isPropertyAccessExpression(callee) && ts.isIdentifier(callee.expression) ? callee.expression.text : null;
				const consumer = calleeRoot !== null ? origins.get(calleeRoot) : undefined;
				if (consumer !== undefined) {
					const seen = new Set<string>();
					const hit = (name: string, op: string): void => {
						// A TABLE symbol passed as an argument is a table access, not a port injection —
						// already counted under kind `table`. Counting it here too produced 1,681 phantom
						// `core-api → db` "DI" edges on the first run, all of them `fn(db, bets)`.
						if (TABLES.has(name)) return;
						const prov = origins.get(name);
						if (prov === undefined || prov.name === consumer.name) return;
						// An imported constant handed to a call is an ordinary import, not an injected port.
						if (CONST_EXPORTS.get(prov.short)?.has(name) === true) {
							diConstantsSuppressed += 1;
							return;
						}
						if (seen.has(`${prov.name}|${name}`)) return;
						seen.add(`${prov.name}|${name}`);
						add({ kind: "di", source: consumer.short, target: prov.short, file: path, line: lineOf(sf, n), symbol: name, op, world });
					};
					const scanArg = (a: ts.Node, depth: number): void => {
						if (depth > 4) return;
						if (ts.isIdentifier(a)) {
							hit(a.text, "wired-in");
							return;
						}
						if (ts.isObjectLiteralExpression(a)) {
							for (const pr of a.properties) {
								if (ts.isPropertyAssignment(pr)) scanArg(pr.initializer, depth + 1);
								else if (ts.isShorthandPropertyAssignment(pr)) hit(pr.name.text, "wired-in (bundle)");
							}
							return;
						}
						if (ts.isArrayLiteralExpression(a)) {
							for (const el of a.elements) scanArg(el, depth + 1);
							return;
						}
						if (ts.isCallExpression(a)) {
							if (ts.isIdentifier(a.expression)) hit(a.expression.text, "wired-in (factory)");
							for (const inner of a.arguments) scanArg(inner, depth + 1);
							return;
						}
						if (ts.isAsExpression(a) || ts.isParenthesizedExpression(a) || ts.isSatisfiesExpression(a)) scanArg(a.expression, depth);
					};
					for (const a of n.arguments) scanArg(a, 0);
				}
			}
		}
		// A RAW DIAL — `fetch("…/api/x")` or a WebSocket url — as opposed to the `apiFetch` seam, which
		// is detected separately above. Matched on the URL literal, because the WebSocket case builds
		// its url in a preceding statement and never passes a literal to the constructor. Three shapes,
		// all narrow on purpose: an argument to `fetch(...)`, an argument to `new WebSocket(...)`, or the
		// initializer of a variable whose name ends in `url`. Anything looser re-counts every `apiFetch`
		// path literal as a second crossing.
		if (ts.isStringLiteralLike(n) || ts.isTemplateExpression(n)) {
			const text = ts.isTemplateExpression(n) ? n.getText(sf) : n.text;
			const m = /\/api\/[A-Za-z0-9\-_/]*/.exec(text);
			if (m !== null) {
				// Walk out through the expression that BUILDS the url, then look at what consumes it.
				let cur: ts.Node = n;
				while (
					cur.parent !== undefined &&
					(ts.isBinaryExpression(cur.parent) || ts.isTemplateExpression(cur.parent) || ts.isTemplateSpan(cur.parent) || ts.isParenthesizedExpression(cur.parent) || ts.isAsExpression(cur.parent))
				) {
					cur = cur.parent;
				}
				const host = cur.parent;
				let shape: string | null = null;
				if (host !== undefined && ts.isCallExpression(host) && ts.isIdentifier(host.expression) && host.expression.text === "fetch") shape = "raw-fetch";
				else if (host !== undefined && ts.isNewExpression(host) && /websocket|wsconstructor/i.test(host.expression.getText(sf))) shape = "raw-websocket";
				else if (host !== undefined && ts.isVariableDeclaration(host) && ts.isIdentifier(host.name) && /url$/i.test(host.name.text)) shape = "raw-dial (url built inline)";
				if (shape !== null) {
					const apiPath = m[0];
					const servedBy = layerServing(apiPath);
					rawApiDials += 1;
					if (servedBy === null) unresolvedApiDials.push({ file: normPath(path), line: lineOf(sf, n), path: apiPath });
					else if (servedBy !== p.layer) {
						add({ kind: "api", source: p.short, target: shortOfLayerName(servedBy), file: path, line: lineOf(sf, n), symbol: apiPath, op: shape, world });
					}
				}
			}
		}
		if (ts.isFunctionDeclaration(n) && n.name !== undefined && /^createMcp(Router|ToolBridge|Session)$/.test(n.name.text)) {
			add({ kind: "mcp", source: p.short, target: "mcp-router", file: path, line: lineOf(sf, n), symbol: n.name.text, op: "router-definition", world });
		}
		if (ts.isPropertyAccessExpression(n) && n.name.text === "router" && /mcp/i.test(n.expression.getText(sf))) {
			add({ kind: "mcp", source: p.short, target: "mcp-router", file: path, line: lineOf(sf, n), symbol: `${n.expression.getText(sf)}.router`, op: "router-host", world });
		}
		ts.forEachChild(n, visit);
	};
	visit(sf);
}

/**

const order = (c: Cell): number => (c.id.startsWith("E.") ? 0 : c.scope === "named" ? 1 : 2);
const matchOrder = [...cells].sort((a, b) => order(a) - order(b));

const cellHits = new Map<string, Inst[]>();
let intraPackage = 0;
let unassigned = 0;
const unassignedSamples: Inst[] = [];

for (const i of insts) {
	const s = byShort.get(i.source);
	const tShort = i.target.replace(/ \[.*$/, "").replace(/^db:.*/, "db");
	const t = byShort.get(tShort);
	if (s !== undefined && t !== undefined && s.name === t.name && i.prodImportsDev !== true) {
		intraPackage += 1;
		continue;
	}
	let hit: Cell | undefined;
	for (const c of matchOrder) {
		if (c.match(i, s, t)) {
			hit = c;
			break;
		}
	}
	if (hit === undefined) {
		unassigned += 1;
		if (unassignedSamples.length < 400) unassignedSamples.push(i);
		continue;
	}
	const arr = cellHits.get(hit.id);
	if (arr === undefined) cellHits.set(hit.id, [i]);
	else arr.push(i);
}

// ---------------------------------------------------------------------------
// rows
// ---------------------------------------------------------------------------

interface Pair {
	label: string;
	prod: number;
	dev: number;
}
interface Row {
	id: string;
	row: "enumerated" | "live";
	scope: "general" | "named";
	matrix: string;
	source: string;
	target: string;
	kind: string;
	/** the PRODUCTION-world verdict */
	verdict: Verdict;
	/** the DEVELOPMENT-world verdict (Ruling 8.2) */
	devVerdict: Verdict;
	/** why the development verdict is what it is */
	devRule: string;
	/** instance counts: `prod` is production-world, `dev` is development-world */
	prod: number;
	dev: number;
	total: number;
	/** for a general row that has named carve-outs: what it would count if they were folded back in */
	coveredIncludingNamed?: number;
	namedCarveOuts?: string[];
	qualifies?: string;
	evidence?: string;
	rule: string;
	alternative: string | null;
	note?: string;
	/** every concrete (source → target) pair the row covers, prod-desc; the replacement for the old per-pair rows */
	pairs: Pair[];
	pairsTotal: number;
	pairsShown: number;
	exTotal: number;
	ex: { s: string; t: string; f: string; l: number; y: string; o: string; k: string }[];
}

const EX_CAP = 24;
const PAIR_CAP = 40;
const toEx = (arr: Inst[]): Row["ex"] =>
	[...arr]
		.sort((a, b) => (a.world === b.world ? 0 : a.world === "production" ? -1 : 1))
		.slice(0, EX_CAP)
		.map((i) => ({ s: i.source, t: i.target, f: i.file, l: i.line, y: i.symbol, o: i.op, k: i.world }));

const tableSuffix = (t: string | undefined): string => {
	if (t === undefined) return "";
	const info = TABLE_INFO.get(t);
	if (info === undefined) return "";
	if (info.state === "owned") return ` · owner: ${info.owner as string}`;
	return " · NO OWNER — violates the completeness rule";
};

const toPairs = (arr: Inst[]): { pairs: Pair[]; total: number } => {
	const m = new Map<string, Pair>();
	for (const i of arr) {
		const label = `${i.source} → ${i.target}${tableSuffix(i.table)}`;
		let p = m.get(label);
		if (p === undefined) {
			p = { label, prod: 0, dev: 0 };
			m.set(label, p);
		}
		if (i.world === "production") p.prod += 1;
		else p.dev += 1;
	}
	const all = [...m.values()].sort((a, b) => b.prod - a.prod || b.dev - a.dev || a.label.localeCompare(b.label));
	return { pairs: all.slice(0, PAIR_CAP), total: all.length };
};

const namedByQualified = new Map<string, string[]>();
for (const c of cells) {
	if (c.scope !== "named" || c.qualifies === undefined) continue;
	const a = namedByQualified.get(c.qualifies) ?? [];
	a.push(c.id);
	namedByQualified.set(c.qualifies, a);
}

/**
 * COLLAPSING THE EMPTY CELLS. Expanding every union into one row per role produces the complete
 * grid, and most of that grid is the default: forbidden with nothing in it. A cell earns a row when
 * it is ALLOWED (the founder wants the permission list complete, unused permissions included), when
 * it has at least one live instance in either world, when it is the enforced rule, or when a named
 * row carves out of it. Everything else is recorded by name in `summary.collapsedCells` with the
 * reason, so nothing vanishes silently — and because every collapsed cell holds zero instances by
 * construction, the collapse cannot move a single count.
 */
const collapsedCells: { id: string; matrix: string; source: string; target: string; kind: string; verdict: Verdict; devVerdict: Verdict; reason: string; rule: string }[] = [];
/**
 * A named row can carry the SAME (source, target, kind) label as the general row it carves out of —
 * `tooling → db:schema-ns [import]` is both. If the general row is left at zero the two are
 * indistinguishable on the page, which is the ambiguity this whole restructure exists to remove. The
 * named row is the rule for that cell, so the emptied general row is collapsed.
 */
const namedTriples = new Set(cells.filter((c) => c.scope === "named").map((c) => `${c.source}|${c.target}|${c.kind}`));
const rows: Row[] = [];
for (const c of cells) {
	const arr = cellHits.get(c.id) ?? [];
	const prod = arr.filter((i) => i.world === "production").length;
	const { pairs, total: pairsTotal } = toPairs(arr);
	const carve = namedByQualified.get(c.id) ?? [];
	const carveTotal = carve.reduce((n, id) => n + (cellHits.get(id)?.length ?? 0), 0);
	const supersededByNamed = c.scope === "general" && arr.length === 0 && namedTriples.has(`${c.source}|${c.target}|${c.kind}`);
	const keep = !supersededByNamed && (c.verdict === "allowed" || arr.length > 0 || carveTotal > 0 || c.pin === true);
	if (!keep) {
		collapsedCells.push({
			id: c.id,
			matrix: c.matrix,
			source: c.source,
			target: c.target,
			kind: c.kind,
			verdict: c.verdict,
			devVerdict: c.devVerdict,
			reason: supersededByNamed
				? "superseded in full by the named row with the same source, target and kind — every instance is the named one"
				: c.verdict === "forbidden"
					? "forbidden by default — nothing written permits it and nothing in the repo does it"
					: "no rule describes it and nothing in the repo does it",
			rule: c.rule,
		});
		continue;
	}
	rows.push({
		id: c.id,
		row: c.scope === "general" ? "enumerated" : "live",
		scope: c.scope,
		matrix: c.matrix,
		source: c.source,
		target: c.target,
		kind: c.kind,
		verdict: c.verdict,
		devVerdict: c.devVerdict,
		devRule: c.devRule,
		prod,
		dev: arr.length - prod,
		total: arr.length,
		...(carve.length > 0 ? { coveredIncludingNamed: arr.length + carveTotal, namedCarveOuts: carve } : {}),
		...(c.qualifies === undefined ? {} : { qualifies: c.qualifies }),
		...(c.evidence === undefined ? {} : { evidence: c.evidence }),
		rule: c.rule,
		alternative: c.alternative,
		...(c.note === undefined ? {} : { note: c.note }),
		pairs,
		pairsTotal,
		pairsShown: pairs.length,
		exTotal: arr.length,
		ex: toEx(arr),
	});
}

// general first; named after, grouped under the general rule they qualify
const generalIndex = new Map(rows.filter((r) => r.scope === "general").map((r, n) => [r.id, n]));
rows.sort((a, b) => {
	if (a.scope !== b.scope) return a.scope === "general" ? -1 : 1;
	if (a.scope === "general") return (generalIndex.get(a.id) ?? 0) - (generalIndex.get(b.id) ?? 0);
	if (a.matrix !== b.matrix) return a.matrix.localeCompare(b.matrix);
	const ga = generalIndex.get(a.qualifies ?? "") ?? 1e9;
	const gb = generalIndex.get(b.qualifies ?? "") ?? 1e9;
	return ga - gb || a.id.localeCompare(b.id);
});

// ---------------------------------------------------------------------------
// reconciliation — the proof that nothing is counted twice
// ---------------------------------------------------------------------------

const generalRows = rows.filter((r) => r.scope === "general");
const namedRows = rows.filter((r) => r.scope === "named");
const sum = (rs: Row[], f: (r: Row) => number): number => rs.reduce((n, r) => n + f(r), 0);
const generalTotal = sum(generalRows, (r) => r.total);
const namedTotal = sum(namedRows, (r) => r.total);
const reconciliation = {
	rawInstances: insts.length,
	intraPackageExcluded: intraPackage,
	boundaryCrossingInstances: insts.length - intraPackage,
	generalRowTotal: generalTotal,
	namedRowTotal: namedTotal,
	matchedNoRow: unassigned,
	sum: generalTotal + namedTotal + intraPackage + unassigned,
	balances: generalTotal + namedTotal + intraPackage + unassigned === insts.length,
	cellsCollapsedAsEmpty: collapsedCells.length,
	instancesInCollapsedCells: 0,
	statement:
		"Every instance is claimed by AT MOST ONE row: matching runs ENFORCED → NAMED → GENERAL and stops at the first hit, so a named row's instances are removed from its general row rather than added on top. generalRowTotal + namedRowTotal + intraPackageExcluded + matchedNoRow must equal rawInstances. Cells collapsed for being empty held zero instances by construction (that is the collapse condition), so the collapse cannot move any count — they are listed by name in `summary.collapsedCells`.",
};
if (!reconciliation.balances) {
	throw new Error(`reconciliation FAILED: ${JSON.stringify(reconciliation)}`);
}
const prodRecon = {
	rawProduction: insts.filter((i) => i.world === "production").length,
	generalProduction: sum(generalRows, (r) => r.prod),
	namedProduction: sum(namedRows, (r) => r.prod),
};

// summary

const tally = (rs: Row[]): Record<string, number> => {
	const o: Record<string, number> = { allowed: 0, forbidden: 0, unsettled: 0 };
	for (const r of rs) o[r.verdict] = (o[r.verdict] ?? 0) + 1;
	return o;
};
/** the same tally for the DEVELOPMENT world, plus the rows where the two worlds disagree */
const devTally = (rs: Row[]): Record<string, number> => {
	const o: Record<string, number> = { allowed: 0, forbidden: 0, unsettled: 0 };
	for (const r of rs) o[r.devVerdict] = (o[r.devVerdict] ?? 0) + 1;
